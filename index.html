App Flutter de Teste

Código completo para receber alertas. Copie e cole no seu projeto Flutter.

pubspec.yaml — dependências 
dependencies:
  flutter:
    sdk: flutter
  firebase_core: ^3.6.0
  firebase_messaging: ^15.1.3
  flutter_local_notifications: ^17.2.2
main.dart — código completo 
import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:firebase_messaging/firebase_messaging.dart';

@pragma('vm:entry-point')
Future<void> bgHandler(RemoteMessage msg) async {
  await Firebase.initializeApp();
}

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp();
  FirebaseMessaging.onBackgroundMessage(bgHandler);
  runApp(const EventoApp());
}

class EventoApp extends StatelessWidget {
  const EventoApp({super.key});
  @override
  Widget build(BuildContext context) => MaterialApp(
    title: 'Alertas Evento',
    theme: ThemeData(
      colorScheme: ColorScheme.fromSeed(
        seedColor: Colors.indigo, brightness: Brightness.dark,
      ),
    ),
    home: const AlertScreen(),
  );
}

class AlertScreen extends StatefulWidget {
  const AlertScreen({super.key});
  @override
  State<AlertScreen> createState() => _AlertScreenState();
}

class _AlertScreenState extends State<AlertScreen> {
  String _token = 'Carregando...';
  final List<String> _msgs = [];

  @override
  void initState() {
    super.initState();
    _setup();
  }

  Future<void> _setup() async {
    final fcm = FirebaseMessaging.instance;

    // Pede permissão (necessário iOS e Android 13+)
    await fcm.requestPermission(
      alert: true, badge: true, sound: true,
    );

    // Inscreve no tópico do evento — recebe todos os alertas
    await fcm.subscribeToTopic('evento2026');
    debugPrint('Inscrito no tópico: evento2026');

    // Obtém o token único deste dispositivo
    final token = await fcm.getToken();
    setState(() => _token = token ?? 'Erro ao obter token');
    debugPrint('FCM Token: $token');

    // Escuta mensagens com o app aberto (foreground)
    FirebaseMessaging.onMessage.listen((RemoteMessage msg) {
      final t = msg.notification?.title ?? '';
      final b = msg.notification?.body  ?? '';
      setState(() => _msgs.insert(0, '$t\n$b'));
    });
  }

  @override
  Widget build(BuildContext context) => Scaffold(
    backgroundColor: const Color(0xFF0A0C1A),
    appBar: AppBar(
      title: const Text('🔔 Alertas do Evento'),
      backgroundColor: const Color(0xFF131630),
    ),
    body: Padding(
      padding: const EdgeInsets.all(16),
      child: Column(children: [
        // ── Token Card ─────────────────────────────────────
        Container(
          width: double.infinity,
          padding: const EdgeInsets.all(14),
          decoration: BoxDecoration(
            color: Colors.indigo.withOpacity(.1),
            border: Border.all(color: Colors.indigo.withOpacity(.3)),
            borderRadius: BorderRadius.circular(14),
          ),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text('Token FCM deste dispositivo:',
                style: TextStyle(color: Colors.white54, fontSize: 11)),
              const SizedBox(height: 6),
              SelectableText(_token,
                style: const TextStyle(
                  fontSize: 10, fontFamily: 'monospace',
                  color: Colors.greenAccent,
                )),
            ],
          ),
        ),
        const SizedBox(height: 16),
        const Align(
          alignment: Alignment.centerLeft,
          child: Text('Alertas Recebidos:',
            style: TextStyle(fontWeight: FontWeight.bold, color: Colors.white)),
        ),
        const SizedBox(height: 8),
        Expanded(
          child: _msgs.isEmpty
            ? const Center(
                child: Text('Aguardando alertas...',
                  style: TextStyle(color: Colors.white30)))
            : ListView.builder(
                itemCount: _msgs.length,
                itemBuilder: (_, i) => Container(
                  margin: const EdgeInsets.only(bottom: 8),
                  padding: const EdgeInsets.all(13),
                  decoration: BoxDecoration(
                    color: Colors.white.withOpacity(.05),
                    borderRadius: BorderRadius.circular(12),
                    border: Border.all(color: Colors.white.withOpacity(.07)),
                  ),
                  child: Text(_msgs[i],
                    style: const TextStyle(
                      fontSize: 13, color: Colors.white70)),
                ),
              ),
        ),
      ]),
    ),
  );
}

📖 Tutorial Completo — FCM HTTP v1

API Atual

Por que FCM HTTP v1?
A API legada (Server Key) foi desativada em junho de 2024. A versão v1 usa OAuth 2.0 + Service Account — mais segura, com suporte a personalizações por plataforma (Android, iOS, Web) e entrega confiável mesmo com o app em segundo plano.

1

Gerar o Service Account Key

→ console.firebase.google.com → projeto alertateste-1a1b4

→ ⚙️ Configurações do projeto → aba Contas de serviço

→ Clique em Gerar nova chave privada → baixe o JSON

→ Cole o conteúdo na aba ⚙️ Configurações deste painel

2

Habilitar Cloud Messaging API

→ Google Cloud Console → APIs → Firebase Cloud Messaging API

→ Clique em Ativar (se ainda não estiver)

→ Confirme que no Firebase Console em Cloud Messaging aparece API Firebase Cloud Messaging (V1) Ativado

3

Endpoint FCM HTTP v1

O endpoint correto para enviar mensagens é:

POST https://fcm.googleapis.com/v1/projects/alertateste-1a1b4/messages:send

Headers obrigatórios:

Authorization: Bearer <ACCESS_TOKEN_OAUTH2> Content-Type: application/json

O token OAuth2 é gerado automaticamente a partir do Service Account JSON.

4

Estrutura do Payload (FCM v1)

Enviar para um tópico:

{ "message": { "topic": "evento2026", "notification": { "title": "🎉 ATENÇÃO!", "body": "Sorteio em 5 minutos no palco!" }, "android": { "priority": "high", "notification": { "channel_id": "event_alerts", "sound": "default", "default_vibrate_timings": true, "notification_priority": "PRIORITY_HIGH", "visibility": "PUBLIC" } }, "apns": { "payload": { "aps": { "sound": "default", "badge": 1 } } } } }

Enviar para um token específico:

{ "message": { "token": "fcm-token-do-dispositivo", "notification": { "title": "...", "body": "..." } } }
5

Configurar App Flutter (build.gradle)

android/build.gradle (nível raiz):

plugins { id("com.google.gms.google-services") version "4.5.0" apply false }

android/app/build.gradle (nível app):

plugins { id("com.android.application") id("com.google.gms.google-services") } dependencies { implementation(platform("com.google.firebase:firebase-bom:34.17.0")) implementation("com.google.firebase:firebase-messaging") }
6

Todos os dispositivos que rodarem o app se inscrevem automaticamente.
